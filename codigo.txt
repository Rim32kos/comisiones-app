import android.app.Application

import android.os.Bundle
import androidx.activity.ComponentActivity
import androidx.activity.compose.setContent

import androidx.compose.runtime.mutableStateOf

import com.comisiones.app.ui.AppNav
import com.comisiones.app.ui.TemaComisiones

class App : Application() {
    override fun onCreate() {
        super.onCreate()
        Alarmas.crearCanales(this)
        Alarmas.programarDiario(this)
    }
}

class MainActivity : ComponentActivity() {
    private val abrirComision = mutableStateOf<Long?>(null)

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        leerIntent(intent)
        setContent {
            TemaComisiones {
                AppNav(abrirComision)
            }
        }
    }

    override fun onNewIntent(intent: Intent) {
        super.onNewIntent(intent)
        leerIntent(intent)
    }

    private fun leerIntent(i: Intent?) {
        val id = i?.getLongExtra(Alarmas.EXTRA_COMISION, -1L) ?: -1L
        if (id > 0) {
            abrirComision.value = id
            i?.removeExtra(Alarmas.EXTRA_COMISION)
        }
    }
}

      (SELECT MAX(m.lastWorkedAt) FROM commissions m WHERE m.clientId = c.id AND m.completed = 0) AS lastWorked

import com.comisiones.app.MainActivity
import com.comisiones.app.R
import com.comisiones.app.data.AppDatabase
import com.comisiones.app.data.Etapas
import com.comisiones.app.util.Ajustes

import com.comisiones.app.util.hoyNumero
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.launch
import java.time.ZonedDateTime

object Alarmas {
    const val CANAL_RECORDATORIOS = "recordatorios"
    const val CANAL_DIARIO = "diario"
    const val EXTRA_COMISION = "comisionId"
    private const val REQ_DIARIO = 999_999
    private const val NOTIF_DIARIO = 999_998
    private const val FLAGS = PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE

    fun crearCanales(ctx: Context) {
        val nm = ctx.getSystemService(NotificationManager::class.java)
        nm.createNotificationChannel(
            NotificationChannel(CANAL_RECORDATORIOS, "Recordatorios de comisiones", NotificationManager.IMPORTANCE_HIGH).apply {
                description = "Los recordatorios que pones en cada comisión"
                enableVibration(true)
            }
        )
        nm.createNotificationChannel(
            NotificationChannel(CANAL_DIARIO, "Resumen diario", NotificationManager.IMPORTANCE_DEFAULT).apply {
                description = "A quién te toca dibujar hoy y qué comisiones van atrasadas"
            }
        )
    }

    private fun programar(ctx: Context, en: Long, pi: PendingIntent) {
        val am = ctx.getSystemService(AlarmManager::class.java)
        try {
            if (Build.VERSION.SDK_INT >= 31 && !am.canScheduleExactAlarms()) {
                am.setAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, en, pi)
            } else {
                am.setExactAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, en, pi)
            }
        } catch (e: SecurityException) {
            am.setAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, en, pi)
        }
    }

    private fun piRecordatorio(ctx: Context, id: Long): PendingIntent =
        PendingIntent.getBroadcast(
            ctx, id.toInt(),
            Intent(ctx, RecordatorioReceiver::class.java).putExtra(EXTRA_COMISION, id),
            FLAGS
        )

    fun programarRecordatorio(ctx: Context, id: Long, en: Long) {
        programar(ctx, en.coerceAtLeast(System.currentTimeMillis() + 3_000), piRecordatorio(ctx, id))
    }

    fun cancelarRecordatorio(ctx: Context, id: Long) {
        ctx.getSystemService(AlarmManager::class.java).cancel(piRecordatorio(ctx, id))
    }

    fun programarDiario(ctx: Context) {
        val aj = Ajustes(ctx)
        val pi = PendingIntent.getBroadcast(ctx, REQ_DIARIO, Intent(ctx, DiarioReceiver::class.java), FLAGS)
        val am = ctx.getSystemService(AlarmManager::class.java)
        am.cancel(pi)
        if (!aj.diarioActivo) return
        val ahora = ZonedDateTime.now()
        var t = ahora.withHour(aj.diarioHora).withMinute(aj.diarioMinuto).withSecond(0).withNano(0)
        if (!t.isAfter(ahora)) t = t.plusDays(1)
        programar(ctx, t.toInstant().toEpochMilli(), pi)
    }

    fun abrirApp(ctx: Context, comisionId: Long?): PendingIntent {
        val i = Intent(ctx, MainActivity::class.java).apply {
            flags = Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_SINGLE_TOP
            if (comisionId != null) putExtra(EXTRA_COMISION, comisionId)
        }
        return PendingIntent.getActivity(ctx, ((comisionId ?: 0L) + 1).toInt(), i, FLAGS)
    }

    fun puedeNotificar(ctx: Context): Boolean =
        Build.VERSION.SDK_INT < 33 ||
            ContextCompat.checkSelfPermission(ctx, Manifest.permission.POST_NOTIFICATIONS) == PackageManager.PERMISSION_GRANTED

    fun notificar(ctx: Context, idNotif: Int, canal: String, titulo: String, texto: String, pi: PendingIntent) {
        if (!puedeNotificar(ctx)) return
        val n = NotificationCompat.Builder(ctx, canal)
            .setSmallIcon(R.drawable.ic_notif)
            .setColor(0xFFFF8FC7.toInt())
            .setContentTitle(titulo)
            .setContentText(texto)
            .setStyle(NotificationCompat.BigTextStyle().bigText(texto))
            .setContentIntent(pi)
            .setAutoCancel(true)
            .setPriority(NotificationCompat.PRIORITY_HIGH)
            .build()
        try {
            NotificationManagerCompat.from(ctx).notify(idNotif, n)
        } catch (e: SecurityException) {
            // sin permiso
        }
    }

    // Resumen diario: a quién toca hoy + comisiones atrasadas.
    suspend fun enviarResumen(ctx: Context) {
        val dao = AppDatabase.get(ctx).dao()
        val rojo = Ajustes(ctx).diasRojo
        val hoy = dao.scheduleForDayNow(hoyNumero()).map { it.clientName }.distinct()
        val atrasadas = dao.activeCommissionsNow().filter { diasDesde(it.commission.lastWorkedAt) >= rojo }
        if (hoy.isEmpty() && atrasadas.isEmpty()) return
        val texto = buildString {
            if (hoy.isNotEmpty()) append("Hoy toca dibujar: ").append(hoy.joinToString(", "))
            if (atrasadas.isNotEmpty()) {
                if (isNotEmpty()) append("\n")
                append("⚠ ${atrasadas.size} comisión(es) sin avance hace $rojo+ días:")
                atrasadas.take(4).forEach {
                    append("\n• ${it.commission.title} (${it.clientName}) — ${diasDesde(it.commission.lastWorkedAt)} días")
                }
            }
        }
        notificar(ctx, NOTIF_DIARIO, CANAL_DIARIO, "🎨 Tu plan de dibujo de hoy", texto, abrirApp(ctx, null))
    }
}

class RecordatorioReceiver : BroadcastReceiver() {
    override fun onReceive(ctx: Context, intent: Intent) {
        val id = intent.getLongExtra(Alarmas.EXTRA_COMISION, -1L)
        if (id < 0) return
        val pendiente = goAsync()
        CoroutineScope(Dispatchers.IO).launch {
            try {
                val dao = AppDatabase.get(ctx).dao()
                val m = dao.commissionNow(id)
                if (m != null) {
                    val cliente = dao.clientNow(m.clientId)?.name ?: ""
                    dao.clearReminder(id)
                    val texto = buildString {
                        append(cliente)
                        if (m.reminderNote.isNotBlank()) append(" — ").append(m.reminderNote)
                        append("\nVas en: ")
                        append(if (m.completed) "Terminada ✓" else Etapas.nombres.getOrElse(m.stage) { "" })
                    }
                    Alarmas.notificar(ctx, id.toInt(), Alarmas.CANAL_RECORDATORIOS, "⏰ ${m.title}", texto, Alarmas.abrirApp(ctx, id))
                }
            } finally {
                pendiente.finish()
            }
        }
    }
}

class DiarioReceiver : BroadcastReceiver() {
    override fun onReceive(ctx: Context, intent: Intent) {
        val pendiente = goAsync()
        CoroutineScope(Dispatchers.IO).launch {
            try {
                Alarmas.enviarResumen(ctx)
            } catch (e: Exception) {
                // ignorar
            } finally {
                Alarmas.programarDiario(ctx)
                pendiente.finish()
            }
        }
    }
}

// Al reiniciar el teléfono o actualizar la app, vuelve a programar todo.
class ArranqueReceiver : BroadcastReceiver() {
    override fun onReceive(ctx: Context, intent: Intent) {
        val pendiente = goAsync()
        CoroutineScope(Dispatchers.IO).launch {
            try {
                Alarmas.crearCanales(ctx)
                Alarmas.programarDiario(ctx)
                AppDatabase.get(ctx).dao().withRemindersNow().forEach { m ->
                    val en = m.reminderAt
                    if (en != null) Alarmas.programarRecordatorio(ctx, m.id, en)
                }
            } finally {
                pendiente.finish()
            }
        }
    }
}

@Composable
fun AppNav(abrirComision: MutableState<Long?>) {
    val vm: AppViewModel = viewModel()
    val nav = rememberNavController()
    val ctx = LocalContext.current
    val snack = remember { SnackbarHostState() }

    val permiso = rememberLauncherForActivityResult(ActivityResultContracts.RequestPermission()) { }
    LaunchedEffect(Unit) {
        if (Build.VERSION.SDK_INT >= 33 && !Alarmas.puedeNotificar(ctx) && !vm.ajustes.permisoPedido) {
            vm.ajustes.permisoPedido = true
            permiso.launch(Manifest.permission.POST_NOTIFICATIONS)
        }
    }
    LaunchedEffect(Unit) { vm.mensajes.collect { snack.showSnackbar(it) } }

    val entrada by nav.currentBackStackEntryAsState()
    val ruta = entrada?.destination?.route

    // Al tocar una notificación, abre directo esa comisión
    val pendiente = abrirComision.value
    val listo = entrada != null
    LaunchedEffect(pendiente, listo) {
        if (pendiente != null && listo) {
            nav.navigate("comision/$pendiente") { launchSingleTop = true }
            abrirComision.value = null
        }
    }

    val secciones = listOf(
        Seccion("clientes", "Clientes", Icons.Default.People),
        Seccion("horario", "Horario", Icons.Default.CalendarMonth),
        Seccion("galeria", "Galería", Icons.Default.PhotoLibrary),
        Seccion("etiquetas", "Etiquetas", Icons.Default.LocalOffer)
    )

    Scaffold(
        containerColor = MaterialTheme.colorScheme.background,
        contentWindowInsets = WindowInsets.navigationBars,
        snackbarHost = { SnackbarHost(snack) },
        bottomBar = {
            if (secciones.any { it.ruta == ruta }) {
                NavigationBar {
                    secciones.forEach { s ->
                        NavigationBarItem(
                            selected = ruta == s.ruta,
                            onClick = {
                                nav.navigate(s.ruta) {
                                    popUpTo(nav.graph.findStartDestination().id) { saveState = true }
                                    launchSingleTop = true
                                    restoreState = true
                                }
                            },
                            icon = { Icon(s.icono, contentDescription = null) },
                            label = { Text(s.nombre, maxLines = 1) }
                        )
                    }
                }
            }
        }
    ) { pad ->
        NavHost(nav, startDestination = "clientes", modifier = Modifier.padding(pad)) {
            composable("clientes") { ClientesScreen(vm, nav) }
            composable("horario") { HorarioScreen(vm, nav) }
            composable("galeria") { GaleriaScreen(vm, nav) }
            composable("etiquetas") { EtiquetasScreen(vm, nav) }
            composable("ajustes") { AjustesScreen(vm, nav) }
            composable("cliente/{id}", arguments = listOf(navArgument("id") { type = NavType.LongType })) {
                ClienteScreen(vm, nav, it.arguments?.getLong("id") ?: 0L)
            }
            composable("comision/{id}", arguments = listOf(navArgument("id") { type = NavType.LongType })) {
                ComisionScreen(vm, nav, it.arguments?.getLong("id") ?: 0L)
            }
            composable("ficha/{id}", arguments = listOf(navArgument("id") { type = NavType.LongType })) {
                FichaScreen(vm, nav, it.arguments?.getLong("id") ?: 0L)
            }
            composable("etiqueta/{id}", arguments = listOf(navArgument("id") { type = NavType.LongType })) {
                EtiquetaScreen(vm, nav, it.arguments?.getLong("id") ?: 0L)
            }
        }
    }

    // Reparte a todos los clientes con comisiones activas entre los días elegidos,
// empezando HOY con quien lleva más días sin ser dibujado. Así nadie se queda atrás.
    fun generarHorarioJusto(dias: List<Int>) = viewModelScope.launch {
        val activos = dao.clientSummariesNow()
            .filter { it.activeCount > 0 }
            .sortedBy { it.lastWorked ?: 0L }
        if (activos.isEmpty()) {
            aviso("No hay clientes con comisiones activas")
            return@launch
        }
        if (dias.isEmpty()) {
            aviso("Elige al menos un día")
            return@launch
        }
        val hoy = hoyNumero()
        val ordenados = dias.sorted()
        val orden = ordenados.filter { it >= hoy } + ordenados.filter { it < hoy }
        val total = maxOf(activos.size, orden.size)
        val posiciones = IntArray(8)
        val nuevos = (0 until total).map { k ->
            val dia = orden[k % orden.size]
            ScheduleSlot(dayOfWeek = dia, clientId = activos[k % activos.size].client.id, position = posiciones[dia]++)
        }
        db.withTransaction {
            dao.clearSchedule()
            nuevos.forEach { dao.insertSlot(it) }
        }
        aviso("Horario justo listo ✨")
    }

fun colorDias(dias: Int, rojo: Int): Color {
    val f = (dias.toFloat() / rojo.coerceAtLeast(1)).coerceIn(0f, 1f)
    return lerp(Color.White, Rojo, f)
}

fun elegirFechaHora(ctx: Context, inicial: Long?, onElegido: (Long) -> Unit) {
    val z = ZoneId.systemDefault()
    val base = inicial?.let { Instant.ofEpochMilli(it).atZone(z) }
        ?: ZonedDateTime.now(z).plusHours(1).withMinute(0)
    DatePickerDialog(
        ctx,
        { _, y, mes, dia ->
            TimePickerDialog(
                ctx,
                { _, h, min ->
                    onElegido(ZonedDateTime.of(y, mes + 1, dia, h, min, 0, 0, z).toInstant().toEpochMilli())
                },
                base.hour, base.minute, true
            ).show()
        },
        base.year, base.monthValue - 1, base.dayOfMonth
    ).show()
}

            listOf(

    borrar?.let { c ->
        DialogoConfirmar(
            "¿Eliminar a ${c.name}?",
            "Se borrarán también todas sus comisiones, referencias e imágenes. No se puede deshacer.",
            onDismiss = { borrar = null }

            borrar = null
            vm.borrarCliente(c)
        }
    }

@Composable
fun ClienteScreen(vm: AppViewModel, nav: NavController, id: Long) {
    val cliente by remember(id) { vm.client(id) }.collectAsState(initial = null)
    val comisiones by remember(id) { vm.commissionsOf(id) }.collectAsState(initial = emptyList())
    var filtro by rememberSaveable { mutableIntStateOf(0) }
    var nueva by remember { mutableStateOf(false) }
    var editar by remember { mutableStateOf(false) }
    var borrar by remember { mutableStateOf(false) }
    val c = cliente

    val activas = comisiones.count { !it.completed }
    val terminadas = comisiones.size - activas
    val lista = comisiones.filter {
        when (filtro) {
            0 -> !it.completed
            1 -> it.completed
            else -> true
        }
    }

    Pantalla(
        c?.name ?: "Cliente",
        onBack = { nav.popBackStack() },
        acciones = {
            if (c != null) {
                IconButton(onClick = { editar = true }) { Icon(Icons.Default.Edit, contentDescription = "Editar") }
                IconButton(onClick = { borrar = true }) { Icon(Icons.Default.Delete, contentDescription = "Eliminar") }
            }
        },
        fab = {
            ExtendedFloatingActionButton(
                onClick = { nueva = true },
                icon = { Icon(Icons.Default.Add, contentDescription = null) },
                text = { Text("Comisión") }
            )
        }
    ) { pad ->
        LazyColumn(
            Modifier.padding(pad).fillMaxSize(),
            contentPadding = PaddingValues(start = 16.dp, end = 16.dp, top = 4.dp, bottom = 96.dp),
            verticalArrangement = Arrangement.spacedBy(10.dp)
        ) {
            if (c != null && (c.contact.isNotBlank() || c.notes.isNotBlank())) {
                item {
                    Card(Modifier.fillMaxWidth()) {
                        Column(Modifier.padding(14.dp), verticalArrangement = Arrangement.spacedBy(4.dp)) {
                            if (c.contact.isNotBlank()) Text("Contacto: ${c.contact}", style = MaterialTheme.typography.bodyMedium)
                            if (c.notes.isNotBlank()) Text(c.notes, color = MaterialTheme.colorScheme.onSurfaceVariant)
                        }
                    }
                }
            }
            item {
                Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                    FilterChip(selected = filtro == 0, onClick = { filtro = 0 }, label = { Text("Activas ($activas)") })
                    FilterChip(selected = filtro == 1, onClick = { filtro = 1 }, label = { Text("Listas ($terminadas)") })
                    FilterChip(selected = filtro == 2, onClick = { filtro = 2 }, label = { Text("Todas") })
                }
            }
            if (lista.isEmpty()) {
                item {
                    Vacio(if (comisiones.isEmpty()) "Este cliente no tiene comisiones.\nToca «+ Comisión»." else "No hay comisiones aquí.")
                }
            }
            items(lista, key = { it.id }) { m ->
                TarjetaComision(m, vm) { nav.navigate("comision/${m.id}") }
            }
        }
    }

    if (nueva) {
        DialogoCampos(
            "Nueva comisión",
            listOf(
                Campo("Título (ej. Miku en un parque)", obligatorio = true),

            ),
            confirmar = "Crear",
            onDismiss = { nueva = false }
        ) { v ->
            nueva = false
            vm.crearComision(id, v[0], v[1]) { nuevoId -> nav.navigate("comision/$nuevoId") }
        }
    }
    if (editar && c != null) DialogoEditarCliente(c, vm) { editar = false }
    if (borrar && c != null) {
        DialogoConfirmar(
            "¿Eliminar a ${c.name}?",
            "Se borrarán también todas sus comisiones, referencias e imágenes. No se puede deshacer.",
            onDismiss = { borrar = false }
        ) {
            borrar = false
            vm.borrarCliente(c)
            nav.popBackStack()
        }
    }
}

    val refs by remember(id) { vm.refsOf(id) }.collectAsState(initial = emptyList())
    val tags by remember(id) { vm.tagsOf(id) }.collectAsState(initial = emptyList())
    val m = comision
    val clienteId = m?.clientId ?: -1L
    val cliente by remember(clienteId) { vm.client(clienteId) }.collectAsState(initial = null)

    var archivo by remember { mutableStateOf<File?>(null) }
    var imagen by remember { mutableStateOf<ImageBitmap?>(null) }
    var cargando by remember { mutableStateOf(true) }
    var visor by remember { mutableStateOf<String?>(null) }

    // Se vuelve a generar sola cada vez que cambia algo de la comisión
    LaunchedEffect(m, refs, tags, cliente) {
        if (m != null) {
            cargando = true
            val f = vm.generarFicha(m, cliente?.name ?: "", refs, tags)
            archivo = f
            imagen = if (f != null) withContext(Dispatchers.IO) {
                BitmapFactory.decodeFile(f.absolutePath)?.asImageBitmap()
            } else null
            cargando = false
        }
    }

    fun compartir() {
        val f = archivo ?: return
        val uri = FileProvider.getUriForFile(ctx, ctx.packageName + ".files", f)
        val envio = Intent(Intent.ACTION_SEND)
            .setType("image/png")
            .putExtra(Intent.EXTRA_STREAM, uri)
            .addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
        ctx.startActivity(Intent.createChooser(envio, "Compartir ficha"))
    }

    Pantalla(
        m?.title ?: "Ficha",
        onBack = { nav.popBackStack() },
        acciones = {
            IconButton(onClick = { compartir() }, enabled = archivo != null) { Icon(Icons.Default.Share, contentDescription = "Compartir") }
            IconButton(
                onClick = { archivo?.let { vm.guardarEnGaleria(it, "ficha_${id}_${System.currentTimeMillis()}.png") } },
                enabled = archivo != null
            ) { Icon(Icons.Default.Download, contentDescription = "Guardar en galería") }
        }
    ) { pad ->
        Column(
            Modifier
                .padding(pad)
                .fillMaxSize()
                .verticalScroll(rememberScrollState())
                .padding(16.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp)
        ) {
            val img = imagen
            if (img != null) {
                Image(
                    bitmap = img,
                    contentDescription = "Ficha en español",
                    contentScale = ContentScale.FillWidth,
                    modifier = Modifier
                        .fillMaxWidth()
                        .clip(RoundedCornerShape(14.dp))
                        .clickable { archivo?.let { visor = it.absolutePath } }
                )
            } else {
                Box(Modifier.fillMaxWidth().height(240.dp), contentAlignment = Alignment.Center) {
                    if (cargando) CircularProgressIndicator() else Text("No se pudo crear la ficha")
                }
            }
            Text(
                "Toca la ficha para verla en grande. Usa ⬇ para guardarla en la galería de tu teléfono o compártela.",
                style = MaterialTheme.typography.bodySmall,
                color = MaterialTheme.colorScheme.onSurfaceVariant

import java.io.File

// Genera una "ficha" en imagen: el pedido traducido al español + etiquetas + las imágenes de referencia.
object Ficha {
    private const val ANCHO = 1080
    private const val MARGEN = 56
    private const val ALTO_MAX = 9000
    private const val SEP = 24

    private val FONDO = Color.parseColor("#17131F")
    private val TARJETA = Color.parseColor("#241D2E")
    private val ROSA = Color.parseColor("#FF8FC7")
    private val LILA = Color.parseColor("#B9A6FF")
    private val BLANCO = Color.parseColor("#F2ECF7")
    private val GRIS = Color.parseColor("#9C93A8")
    private val VERDE = Color.parseColor("#5BD68A")
    private val AMBAR = Color.parseColor("#FFC46B")

    private class Bloque(val alto: Int, val dibujar: (Canvas, Float) -> Unit)

    fun archivoDe(ctx: Context, id: Long): File = File(File(ctx.filesDir, "fichas").apply { mkdirs() }, "ficha_$id.png")

    suspend fun generar(
        ctx: Context,
        m: Commission,
        cliente: String,
        refs: List<Reference>,
        tags: List<Tag>
    ): File = withContext(Dispatchers.IO) {
        val ancho = ANCHO - MARGEN * 2
        val bloques = mutableListOf<Bloque>()
        val bitmaps = mutableListOf<Bitmap>()

        fun texto(t: String, tam: Float, color: Int, negrita: Boolean = false, despues: Int = 12) {
            val l = layout(t, tam, color, negrita, ancho)
            bloques += Bloque(l.height + despues) { c, y ->
                c.save(); c.translate(MARGEN.toFloat(), y); l.draw(c); c.restore()
            }
        }

        fun separador() {
            bloques += Bloque(SEP * 2) { c, y ->
                val p = Paint(Paint.ANTI_ALIAS_FLAG).apply { color = TARJETA; strokeWidth = 4f }
                c.drawLine(MARGEN.toFloat(), y + SEP, (ANCHO - MARGEN).toFloat(), y + SEP, p)
            }
        }

        // Encabezado
        texto("✦ FICHA DE COMISIÓN", 30f, ROSA, true, 14)
        texto(m.title, 66f, BLANCO, true, 8)
        texto("Cliente: $cliente", 40f, LILA, false, 18)
        val estado = if (m.completed) "✓ Terminada"
        else "Etapa actual: ${Etapas.nombres.getOrElse(m.stage) { "" }} (${m.stage + 1}/4)"
        texto(estado, 36f, if (m.completed) VERDE else AMBAR, true, 10)

        // Etiquetas
        TipoEtiqueta.nombres.forEachIndexed { tipo, nombre ->
            val deTipo = tags.filter { it.type == tipo }
            if (deTipo.isNotEmpty()) texto("$nombre: " + deTipo.joinToString(" · ") { it.name }, 36f, BLANCO, false, 8)
        }
        separador()

        // Pedido
        texto("LO QUE PIDIÓ (EN ESPAÑOL)", 30f, ROSA, true, 14)
        val desc = when {
            m.descriptionEs.isNotBlank() -> m.descriptionEs
            m.descriptionEn.isNotBlank() -> m.descriptionEn + "\n(todavía sin traducir)"
            else -> "(sin descripción)"
        }
        texto(desc, 42f, BLANCO, false, 16)

        val textos = refs.filter { it.kind == 1 }
        if (textos.isNotEmpty()) {
            separador()
            texto("NOTAS DE REFERENCIA", 30f, ROSA, true, 14)
            textos.forEach { r -> texto("• " + r.textEs.ifBlank { r.textEn }, 38f, BLANCO, false, 14) }
        }

        // Imágenes en cuadrícula de 2 columnas
        val imagenes = refs.filter { it.kind == 0 }.take(10)
        if (imagenes.isNotEmpty()) {
            separador()
            texto("IMÁGENES DE REFERENCIA", 30f, ROSA, true, 18)
            val hueco = 20
            val columnas = if (imagenes.size == 1) 1 else 2
            val celda = (ancho - hueco * (columnas - 1)) / columnas
            val altoMaxCelda = if (columnas == 1) 1300 else 760
            imagenes.chunked(columnas).forEach { fila ->
                val bms = fila.mapNotNull { Archivos.decodificar(it.imagePath, celda) }
                bitmaps += bms
                if (bms.isNotEmpty()) {
                    val altos = bms.map { b -> (b.height * (celda.toFloat() / b.width)).toInt().coerceAtMost(altoMaxCelda) }
                    val altoFila = altos.max()
                    bloques += Bloque(altoFila + hueco) { c, y ->
                        bms.forEachIndexed { i, b ->
                            val x = MARGEN + i * (celda + hueco)
                            val destino = RectF(x.toFloat(), y, (x + celda).toFloat(), y + altos[i])
                            val clip = Path().apply { addRoundRect(destino, 22f, 22f, Path.Direction.CW) }
                            // Recorta la imagen si es demasiado alta (se queda con el centro)
                            val altoFuente = (altos[i] * (b.width.toFloat() / celda)).toInt().coerceAtMost(b.height)
                            val top = (b.height - altoFuente) / 2
                            c.save(); c.clipPath(clip)
                            c.drawBitmap(b, Rect(0, top, b.width, top + altoFuente), destino, Paint(Paint.FILTER_BITMAP_FLAG))
                            c.restore()
                        }
                    }
                }
            }
        }

        texto("Mis Comisiones · ${formatoDia(System.currentTimeMillis())}", 26f, GRIS, false, 0)

        val alto = (MARGEN * 2 + bloques.sumOf { it.alto }).coerceAtMost(ALTO_MAX)
        val bmp = Bitmap.createBitmap(ANCHO, alto, Bitmap.Config.ARGB_8888)
        val canvas = Canvas(bmp)
        canvas.drawColor(FONDO)
        // franja de color arriba
        canvas.drawRect(0f, 0f, ANCHO.toFloat(), 14f, Paint().apply { color = ROSA })
        var y = MARGEN.toFloat()
        for (b in bloques) {
            if (y > alto) break
            b.dibujar(canvas, y)
            y += b.alto
        }

        val archivo = archivoDe(ctx, m.id)
        archivo.outputStream().use { bmp.compress(Bitmap.CompressFormat.PNG, 100, it) }
        bmp.recycle()
        bitmaps.forEach { it.recycle() }
        archivo
    }

    private fun layout(t: String, tam: Float, color: Int, negrita: Boolean, ancho: Int): StaticLayout {
        val p = TextPaint(Paint.ANTI_ALIAS_FLAG).apply {
            textSize = tam
            this.color = color
            typeface = if (negrita) Typeface.DEFAULT_BOLD else Typeface.DEFAULT
        }
        return StaticLayout.Builder.obtain(t, 0, t.length, p, ancho)
            .setAlignment(Layout.Alignment.ALIGN_NORMAL)
            .setLineSpacing(0f, 1.15f)
            .build()
    }
}

fun diasDesde(t: Long?): Int {
    if (t == null) return 0
    val z = ZoneId.systemDefault()
    val d = ChronoUnit.DAYS.between(Instant.ofEpochMilli(t).atZone(z).toLocalDate(), LocalDate.now(z))
    return d.toInt().coerceAtLeast(0)
}

// ---------------- Traductor (inglés → español, funciona sin internet tras descargar) ----------------

object Traductor {
    private val traductor by lazy {
        Translation.getClient(
            TranslatorOptions.Builder()
                .setSourceLanguage(TranslateLanguage.ENGLISH)
                .setTargetLanguage(TranslateLanguage.SPANISH)
                .build()
        )
    }

    suspend fun descargar() {
        esperar(traductor.downloadModelIfNeeded(DownloadConditions.Builder().build()))
    }

    suspend fun traducir(texto: String): String {
        if (texto.isBlank()) return ""
        descargar()
        // Línea por línea para conservar los saltos de línea del cliente
        return texto.split("\n").withIndex().fold("") { acc, (i, linea) -> acc + (if (i == 0) "" else "\n") +
            if (linea.isBlank()) linea else esperar(traductor.translate(linea))
        }
    }

    private suspend fun <T> esperar(tarea: Task<T>): T = suspendCancellableCoroutine { cont ->
        tarea.addOnSuccessListener { r -> if (cont.isActive) cont.resume(r) }
        tarea.addOnFailureListener { e -> if (cont.isActive) cont.resumeWithException(e) }
    }
}

object Archivos {
    // Copia una imagen elegida a la memoria interna de la app para que no se pierda.
    fun copiarImagen(ctx: Context, uri: Uri): String? = try {
        val dir = File(ctx.filesDir, "refs").apply { mkdirs() }
        val f = File(dir, UUID.randomUUID().toString() + ".img")
        val ok = ctx.contentResolver.openInputStream(uri)?.use { entrada ->
            f.outputStream().use { salida -> entrada.copyTo(salida) }
            true
        } ?: false
        if (ok) f.absolutePath else null
    } catch (e: Exception) {
        null
    }

    fun borrar(path: String) {
        if (path.isNotBlank()) runCatching { File(path).delete() }
    }

    // Decodifica una imagen reduciéndola al ancho indicado (respeta la rotación de la foto).
    fun decodificar(path: String, anchoMax: Int): Bitmap? = try {
        val src = ImageDecoder.createSource(File(path))
        ImageDecoder.decodeBitmap(src) { decoder, info, _ ->
            decoder.allocator = ImageDecoder.ALLOCATOR_SOFTWARE
            val w = info.size.width
            val h = info.size.height
            if (w > anchoMax && w > 0) {
                val escala = anchoMax.toFloat() / w
                decoder.setTargetSize(anchoMax, (h * escala).toInt().coerceAtLeast(1))
            }
        }
    } catch (e: Exception) {
        null
    }

    // Guarda una copia en la galería del teléfono (Imágenes/MisComisiones).
    fun guardarEnGaleria(ctx: Context, archivo: File, nombre: String): Boolean = try {
        val resolver = ctx.contentResolver
        val valores = ContentValues().apply {
            put(MediaStore.Images.Media.DISPLAY_NAME, nombre)
            put(MediaStore.Images.Media.MIME_TYPE, "image/png")
            put(MediaStore.Images.Media.RELATIVE_PATH, Environment.DIRECTORY_PICTURES + "/MisComisiones")
            put(MediaStore.Images.Media.IS_PENDING, 1)
        }
        val uri = resolver.insert(MediaStore.Images.Media.EXTERNAL_CONTENT_URI, valores)
        if (uri == null) {
            false
        } else {
            resolver.openOutputStream(uri)?.use { salida -> archivo.inputStream().use { it.copyTo(salida) } }
            valores.clear()
            valores.put(MediaStore.Images.Media.IS_PENDING, 0)
            resolver.update(uri, valores, null, null)
            true
        }
    } catch (e: Exception) {
        false
    }
}
