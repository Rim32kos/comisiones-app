package comisiones

import androidx.compose.runtime.mutableStateMapOf
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.launch
import java.io.File

// Todo lo que se puede hacer en la app (crear, editar, borrar, traducir...)
object Acciones {
    private val alcance = CoroutineScope(SupervisorJob() + Dispatchers.IO)

    // Claves "d<id>" (descripción) o "r<id>" (referencia) mientras se traducen
    val traduciendo = mutableStateMapOf<String, Boolean>()

    private fun d(): Datos = Almacen.datos.value

    private fun cambiarComision(id: Long, f: (Comision) -> Comision) =
        Almacen.cambiar { d -> d.copy(comisiones = d.comisiones.map { if (it.id == id) f(it) else it }) }

    private fun cambiarRef(id: Long, f: (Referencia) -> Referencia) =
        Almacen.cambiar { d -> d.copy(referencias = d.referencias.map { if (it.id == id) f(it) else it }) }

    // ---------- Clientes ----------
    fun guardarCliente(id: Long?, nombre: String, contacto: String, notas: String) {
        if (nombre.isBlank()) return
        if (id == null) {
            Almacen.cambiar { d ->
                d.copy(
                    sigId = d.sigId + 1,
                    clientes = d.clientes + Cliente(d.sigId, nombre.trim(), contacto.trim(), notas.trim(), ahora())
                )
            }
            Avisos.decir("Cliente agregado")
        } else {
            Almacen.cambiar { d ->
                d.copy(clientes = d.clientes.map {
                    if (it.id == id) it.copy(nombre = nombre.trim(), contacto = contacto.trim(), notas = notas.trim()) else it
                })
            }
        }
    }

    fun borrarCliente(id: Long) {
        val ids = d().comisiones.filter { it.clienteId == id }.map { it.id }.toSet()
        val rutas = d().referencias.filter { it.comisionId in ids && it.tipo == 0 }.map { it.ruta }
        Almacen.cambiar { d ->
            d.copy(
                clientes = d.clientes.filter { it.id != id },
                comisiones = d.comisiones.filter { it.clienteId != id },
                referencias = d.referencias.filter { it.comisionId !in ids },
                horario = d.horario.filter { it.clienteId != id }
            )
        }
        rutas.forEach { Archivos.borrar(it) }
        ids.forEach { runCatching { Ficha.archivoDe(it).delete() } }
        Avisos.decir("Cliente eliminado")
    }

    // ---------- Comisiones ----------
    fun crearComision(clienteId: Long, titulo: String, descEn: String): Long {
        val t = ahora()
        val id = Almacen.cambiarY { d ->
            val nuevo = d.sigId
            d.copy(
                sigId = nuevo + 1,
                comisiones = d.comisiones + Comision(
                    id = nuevo, clienteId = clienteId, titulo = titulo.trim(), descEn = descEn.trim(),
                    creada = t, ultimoTrabajo = t
                )
            ) to nuevo
        }
        if (descEn.isNotBlank()) traducirDescripcion(id)
        return id
    }

    fun editarComision(id: Long, titulo: String, descEn: String) {
        val m = d().comision(id) ?: return
        val cambio = descEn.trim() != m.descEn
        cambiarComision(id) {
            it.copy(titulo = titulo.trim(), descEn = descEn.trim(), descEs = if (cambio) "" else it.descEs)
        }
        if (cambio && descEn.isNotBlank()) traducirDescripcion(id)
    }

    fun borrarComision(id: Long) {
        val rutas = d().referencias.filter { it.comisionId == id && it.tipo == 0 }.map { it.ruta }
        Almacen.cambiar { d ->
            d.copy(
                comisiones = d.comisiones.filter { it.id != id },
                referencias = d.referencias.filter { it.comisionId != id }
            )
        }
        rutas.forEach { Archivos.borrar(it) }
        runCatching { Ficha.archivoDe(id).delete() }
        Avisos.decir("Comisión eliminada")
    }

    fun setTerminada(id: Long, si: Boolean) {
        val t = ahora()
        cambiarComision(id) {
            if (si) it.copy(terminada = true, terminadaEn = t, ultimoTrabajo = t, etapa = 3, recordatorio = null)
            else it.copy(terminada = false, terminadaEn = null, ultimoTrabajo = t)
        }
        if (si) Avisos.decir("¡Comisión terminada! ✓")
    }

    fun setEtapa(id: Long, etapa: Int) =
        cambiarComision(id) { it.copy(etapa = etapa, terminada = false, terminadaEn = null, ultimoTrabajo = ahora()) }

    fun dibujeHoy(id: Long) {
        cambiarComision(id) { it.copy(ultimoTrabajo = ahora()) }
        Avisos.decir("¡Anotado! El contador de días volvió a 0")
    }

    fun ponerRecordatorio(id: Long, en: Long, nota: String) {
        cambiarComision(id) { it.copy(recordatorio = en, notaRecordatorio = nota.trim()) }
        Avisos.decir("Recordatorio: " + formatoFecha(en))
    }

    fun quitarRecordatorio(id: Long) = cambiarComision(id) { it.copy(recordatorio = null) }

    // ---------- Traducción ----------
    fun traducirDescripcion(id: Long) {
        val clave = "d$id"
        if (traduciendo[clave] == true) return
        traduciendo[clave] = true
        alcance.launch {
            try {
                val en = d().comision(id)?.descEn ?: ""
                val es = Traductor.traducir(en)
                cambiarComision(id) { it.copy(descEs = es) }
            } catch (e: Exception) {
                Avisos.decir("No se pudo traducir. Revisa tu conexión a internet.")
            } finally {
                traduciendo.remove(clave)
            }
        }
    }

    fun traducirRef(id: Long) {
        val clave = "r$id"
        if (traduciendo[clave] == true) return
        traduciendo[clave] = true
        alcance.launch {
            try {
                val en = d().referencias.firstOrNull { it.id == id }?.textoEn ?: ""
                val es = Traductor.traducir(en)
                cambiarRef(id) { it.copy(textoEs = es) }
            } catch (e: Exception) {
                Avisos.decir("No se pudo traducir. Revisa tu conexión a internet.")
            } finally {
                traduciendo.remove(clave)
            }
        }
    }

    // ---------- Referencias ----------
    fun agregarTexto(comisionId: Long, textoEn: String) {
        if (textoEn.isBlank()) return
        val t = ahora()
        val id = Almacen.cambiarY { d ->
            val nuevo = d.sigId
            d.copy(sigId = nuevo + 1, referencias = d.referencias + Referencia(nuevo, comisionId, 1, textoEn = textoEn.trim(), creada = t)) to nuevo
        }
        traducirRef(id)
    }

    fun editarTexto(refId: Long, textoEn: String) {
        cambiarRef(refId) { it.copy(textoEn = textoEn.trim(), textoEs = "") }
        traducirRef(refId)
    }

    private fun agregarRutas(comisionId: Long, rutas: List<String>) {
        if (rutas.isEmpty()) {
            Avisos.decir("No encontré imágenes para agregar")
            return
        }
        val t = ahora()
        Almacen.cambiar { d ->
            var s = d.sigId
            val nuevas = rutas.mapIndexed { i, r -> Referencia(s++, comisionId, 0, ruta = r, creada = t + i) }
            d.copy(sigId = s, referencias = d.referencias + nuevas)
        }
        Avisos.decir(if (rutas.size == 1) "Imagen agregada" else "${rutas.size} imágenes agregadas")
    }

    fun agregarArchivos(comisionId: Long, archivos: List<File>) {
        alcance.launch { agregarRutas(comisionId, archivos.filter { Archivos.esImagen(it) }.mapNotNull { Archivos.copiar(it) }) }
    }

    fun pegarDelPortapapeles(comisionId: Long) {
        val rutas = Archivos.leerPortapapeles()
        if (rutas.isEmpty()) Avisos.decir("No hay ninguna imagen copiada. Copia una imagen (clic derecho → Copiar imagen) y vuelve a intentar.")
        else agregarRutas(comisionId, rutas)
    }

    fun borrarRef(id: Long) {
        val r = d().referencias.firstOrNull { it.id == id } ?: return
        Almacen.cambiar { d -> d.copy(referencias = d.referencias.filter { it.id != id }) }
        if (r.tipo == 0) {
            Imagenes.olvidar(r.ruta)
            Archivos.borrar(r.ruta)
        }
    }

    // ---------- Etiquetas ----------
    fun agregarEtiqueta(comisionId: Long, nombre: String, tipo: Int) {
        val limpio = nombre.trim().replace(Regex("\\s+"), " ")
        if (limpio.isEmpty()) return
        val clave = limpio.lowercase()
        Almacen.cambiar { d ->
            val existente = d.etiquetas.firstOrNull { it.clave == clave && it.tipo == tipo }
            val et = existente ?: Etiqueta(d.sigId, limpio, clave, tipo)
            val base = if (existente == null) d.copy(sigId = d.sigId + 1, etiquetas = d.etiquetas + et) else d
            base.copy(comisiones = base.comisiones.map {
                if (it.id == comisionId && et.id !in it.etiquetas) it.copy(etiquetas = it.etiquetas + et.id) else it
            })
        }
    }

    fun quitarEtiqueta(comisionId: Long, etiquetaId: Long) =
        cambiarComision(comisionId) { it.copy(etiquetas = it.etiquetas - etiquetaId) }

    fun borrarEtiqueta(etiquetaId: Long) {
        Almacen.cambiar { d ->
            d.copy(
                etiquetas = d.etiquetas.filter { it.id != etiquetaId },
                comisiones = d.comisiones.map { it.copy(etiquetas = it.etiquetas - etiquetaId) }
            )
        }
    }

    // ---------- Horario ----------
    fun agregarAlHorario(dia: Int, clienteId: Long) {
        Almacen.cambiar { d ->
            val pos = d.horario.count { it.dia == dia }
            d.copy(sigId = d.sigId + 1, horario = d.horario + Turno(d.sigId, dia, clienteId, pos))
        }
    }

    fun quitarDelHorario(id: Long) = Almacen.cambiar { d -> d.copy(horario = d.horario.filter { it.id != id }) }

    // Reparte a todos los clientes con comisiones activas entre los días elegidos,
    // empezando HOY con quien lleva más días sin ser dibujado. Así nadie se queda atrás.
    fun generarHorarioJusto(dias: List<Int>) {
        val activos = d().resumenes().filter { it.activas > 0 }.sortedBy { it.ultimo ?: 0L }
        if (activos.isEmpty()) {
            Avisos.decir("No hay clientes con comisiones activas")
            return
        }
        if (dias.isEmpty()) {
            Avisos.decir("Elige al menos un día")
            return
        }
        val hoy = hoyNumero()
        val ordenados = dias.sorted()
        val orden = ordenados.filter { it >= hoy } + ordenados.filter { it < hoy }
        val total = maxOf(activos.size, orden.size)
        Almacen.cambiar { d ->
            var s = d.sigId
            val posiciones = IntArray(8)
            val nuevos = (0 until total).map { k ->
                val dia = orden[k % orden.size]
                Turno(s++, dia, activos[k % activos.size].cliente.id, posiciones[dia]++)
            }
            d.copy(sigId = s, horario = nuevos)
        }
        Avisos.decir("Horario justo listo ✨")
    }

    // ---------- Ajustes ----------
    fun cambiarAjustes(f: (Ajustes) -> Ajustes) = Almacen.cambiar { d -> d.copy(ajustes = f(d.ajustes)) }
}

// ---------------- Consultas ----------------

data class ResumenCliente(val cliente: Cliente, val activas: Int, val total: Int, val ultimo: Long?)

fun Datos.cliente(id: Long): Cliente? = clientes.firstOrNull { it.id == id }
fun Datos.comision(id: Long): Comision? = comisiones.firstOrNull { it.id == id }
fun Datos.etiqueta(id: Long): Etiqueta? = etiquetas.firstOrNull { it.id == id }

fun Datos.resumenes(): List<ResumenCliente> =
    clientes.sortedBy { it.nombre.lowercase() }.map { c ->
        val todas = comisiones.filter { it.clienteId == c.id }
        val activas = todas.filter { !it.terminada }
        ResumenCliente(c, activas.size, todas.size, activas.maxOfOrNull { it.ultimoTrabajo })
    }

fun Datos.comisionesDe(clienteId: Long): List<Comision> =
    comisiones.filter { it.clienteId == clienteId }
        .sortedWith(compareBy<Comision> { it.terminada }.thenByDescending { it.creada })

fun Datos.refsDe(comisionId: Long): List<Referencia> =
    referencias.filter { it.comisionId == comisionId }.sortedBy { it.creada }

fun Datos.etiquetasDe(m: Comision): List<Etiqueta> =
    m.etiquetas.mapNotNull { etiqueta(it) }.sortedWith(compareBy<Etiqueta> { it.tipo }.thenBy { it.nombre.lowercase() })

fun Datos.usos(etiquetaId: Long): Int = comisiones.count { etiquetaId in it.etiquetas }

fun Datos.conteoEtiquetas(): List<Pair<Etiqueta, Int>> =
    etiquetas.map { it to usos(it.id) }
        .sortedWith(compareByDescending<Pair<Etiqueta, Int>> { it.second }.thenBy { it.first.nombre.lowercase() })

fun Datos.textoResumen(): String? {
    val rojo = ajustes.diasRojo
    val hoy = horario.filter { it.dia == hoyNumero() }.sortedBy { it.pos }
        .mapNotNull { cliente(it.clienteId)?.nombre }.distinct()
    val atrasadas = comisiones.filter { !it.terminada && diasDesde(it.ultimoTrabajo) >= rojo }
        .sortedBy { it.ultimoTrabajo }
    if (hoy.isEmpty() && atrasadas.isEmpty()) return null
    return buildString {
        if (hoy.isNotEmpty()) append("Hoy toca dibujar: ").append(hoy.joinToString(", "))
        if (atrasadas.isNotEmpty()) {
            if (isNotEmpty()) append("\n")
            append("${atrasadas.size} comisión(es) sin avance hace $rojo+ días:")
            atrasadas.take(4).forEach {
                append("\n• ${it.titulo} (${cliente(it.clienteId)?.nombre ?: ""}) — ${diasDesde(it.ultimoTrabajo)} días")
            }
        }
    }
}

object Almacen {
    val carpeta: File = File(System.getenv("APPDATA") ?: System.getProperty("user.home"), "MisComisiones").apply { mkdirs() }
    val carpetaRefs: File = File(carpeta, "referencias").apply { mkdirs() }
    val carpetaFichas: File = File(carpeta, "fichas").apply { mkdirs() }
    private val carpetaRespaldos: File = File(carpeta, "respaldos").apply { mkdirs() }
    private val archivo = File(carpeta, "datos.json")
    private val json = Json {
        ignoreUnknownKeys = true
        encodeDefaults = true
        prettyPrint = true
    }
    private val escritor = Executors.newSingleThreadExecutor { r -> Thread(r, "guardar-datos") }
    private val estado = MutableStateFlow(cargar())
    val datos: StateFlow<Datos> = estado.asStateFlow()

    private fun cargar(): Datos =
        try {
            if (archivo.exists()) json.decodeFromString(Datos.serializer(), archivo.readText(Charsets.UTF_8)) else Datos()
        } catch (e: Exception) {
            runCatching { archivo.copyTo(File(carpeta, "datos_danado_${System.currentTimeMillis()}.json"), overwrite = true) }
            Datos()
        }

    private fun guardar(d: Datos) {
        try {
            // Un respaldo por día (se guardan los últimos 14)
            if (archivo.exists()) {
                val hoy = File(carpetaRespaldos, "datos_${LocalDate.now()}.json")
                if (!hoy.exists()) {
                    archivo.copyTo(hoy)
                    carpetaRespaldos.listFiles()?.sortedBy { it.name }?.dropLast(14)?.forEach { it.delete() }
                }
            }
            val tmp = File(carpeta, "datos.tmp")
            tmp.writeText(json.encodeToString(Datos.serializer(), d), Charsets.UTF_8)
            try {
                Files.move(tmp.toPath(), archivo.toPath(), StandardCopyOption.REPLACE_EXISTING, StandardCopyOption.ATOMIC_MOVE)
            } catch (e: Exception) {
                Files.move(tmp.toPath(), archivo.toPath(), StandardCopyOption.REPLACE_EXISTING)
            }
        } catch (e: Exception) {
            Avisos.decir("No se pudo guardar: ${e.message}")
        }
    }

    fun <T> cambiarY(f: (Datos) -> Pair<Datos, T>): T = synchronized(this) {
        val (nuevo, r) = f(estado.value)
        estado.value = nuevo
        escritor.execute { guardar(nuevo) }
        r
    }

    fun cambiar(f: (Datos) -> Datos) {
        cambiarY { d -> f(d) to Unit }
    }

    fun terminarDeGuardar() {
        escritor.shutdown()
        runCatching { escritor.awaitTermination(5, java.util.concurrent.TimeUnit.SECONDS) }
    }
}

object Avisos {
    val flujo = MutableSharedFlow<String>(extraBufferCapacity = 16)
    fun decir(t: String) {
        flujo.tryEmit(t)
    }
}

    fun generar(m: Comision, cliente: String, refs: List<Referencia>, tags: List<Etiqueta>): File {
        val ancho = ANCHO - MARGEN * 2
        val medir = BufferedImage(1, 1, BufferedImage.TYPE_INT_ARGB).createGraphics()
        suavizar(medir)
        val bloques = mutableListOf<Bloque>()

        fun fuente(tam: Int, negrita: Boolean) = Font("Segoe UI", if (negrita) Font.BOLD else Font.PLAIN, tam)

        fun envolver(t: String, f: Font): List<String> {
            val fm = medir.getFontMetrics(f)
            val res = mutableListOf<String>()
            for (parrafo in t.split("\n")) {
                if (parrafo.isBlank()) {
                    res += ""
                    continue
                }
                var linea = ""
                for (palabra in parrafo.split(" ")) {
                    var p = palabra
                    val prueba = if (linea.isEmpty()) p else "$linea $p"
                    if (fm.stringWidth(prueba) <= ancho) {
                        linea = prueba
                    } else {
                        if (linea.isNotEmpty()) res += linea
                        while (p.length > 1 && fm.stringWidth(p) > ancho) {
                            var k = p.length
                            while (k > 1 && fm.stringWidth(p.substring(0, k)) > ancho) k--
                            res += p.substring(0, k)
                            p = p.substring(k)
                        }
                        linea = p
                    }
                }
                res += linea
            }
            return res
        }

        fun texto(t: String, tam: Int, color: Color, negrita: Boolean = false, despues: Int = 12) {
            val f = fuente(tam, negrita)
            val fm = medir.getFontMetrics(f)
            val lineas = envolver(t, f)
            val altoLinea = (fm.height * 1.15).toInt()
            bloques += Bloque(lineas.size * altoLinea + despues) { g, y ->
                g.font = f
                g.color = color
                lineas.forEachIndexed { i, l -> g.drawString(l, MARGEN, y + i * altoLinea + fm.ascent) }
            }
        }

        fun separador() {
            bloques += Bloque(SEP * 2) { g, y ->
                g.color = LINEA
                g.fillRect(MARGEN, y + SEP - 2, ANCHO - MARGEN * 2, 4)
            }
        }

        // Encabezado
        texto("FICHA DE COMISIÓN", 28, ROSA, true, 14)
        texto(m.titulo, 60, BLANCO, true, 8)
        texto("Cliente: $cliente", 38, LILA, false, 18)
        val estado = if (m.terminada) "Terminada"
        else "Etapa actual: ${Etapas.nombres.getOrElse(m.etapa) { "" }} (${m.etapa + 1}/4)"
        texto(estado, 34, if (m.terminada) VERDE else AMBAR, true, 10)

        // Etiquetas
        TipoEtiqueta.nombres.forEachIndexed { tipo, nombre ->
            val deTipo = tags.filter { it.tipo == tipo }
            if (deTipo.isNotEmpty()) texto("$nombre: " + deTipo.joinToString(" · ") { it.nombre }, 34, BLANCO, false, 8)
        }
        separador()

        // Pedido
        texto("LO QUE PIDIÓ (EN ESPAÑOL)", 28, ROSA, true, 14)
        val desc = when {
            m.descEs.isNotBlank() -> m.descEs
            m.descEn.isNotBlank() -> m.descEn + "\n(todavía sin traducir)"
            else -> "(sin descripción)"
        }
        texto(desc, 40, BLANCO, false, 16)

        val textos = refs.filter { it.tipo == 1 }
        if (textos.isNotEmpty()) {
            separador()
            texto("NOTAS DE REFERENCIA", 28, ROSA, true, 14)
            textos.forEach { r -> texto("• " + r.textoEs.ifBlank { r.textoEn }, 36, BLANCO, false, 14) }
        }

        // Imágenes en cuadrícula de 2 columnas
        val imagenes = refs.filter { it.tipo == 0 }.take(10)
        if (imagenes.isNotEmpty()) {
            separador()
            texto("IMÁGENES DE REFERENCIA", 28, ROSA, true, 18)
            val hueco = 20
            val columnas = if (imagenes.size == 1) 1 else 2
            val celda = (ancho - hueco * (columnas - 1)) / columnas
            val altoMaxCelda = if (columnas == 1) 1300 else 760
            imagenes.chunked(columnas).forEach { fila ->
                val bms = fila.mapNotNull { r ->
                    Imagenes.cargarSkia(r.ruta)?.let { Imagenes.aBuffered(Imagenes.escalada(it, celda)) }
                }
                if (bms.isNotEmpty()) {
                    val altos = bms.map { b -> (b.height * (celda.toFloat() / b.width)).toInt().coerceAtMost(altoMaxCelda) }
                    val altoFila = altos.max()
                    bloques += Bloque(altoFila + hueco) { g, y ->
                        bms.forEachIndexed { i, b ->
                            val x = MARGEN + i * (celda + hueco)
                            val alto = altos[i]
                            // Si la imagen es muy alta se recorta y se queda con el centro
                            val altoFuente = (alto * (b.width.toFloat() / celda)).toInt().coerceAtMost(b.height)
                            val arriba = (b.height - altoFuente) / 2
                            val anterior = g.clip
                            g.clip = RoundRectangle2D.Float(x.toFloat(), y.toFloat(), celda.toFloat(), alto.toFloat(), 44f, 44f)
                            g.drawImage(b, x, y, x + celda, y + alto, 0, arriba, b.width, arriba + altoFuente, null)
                            g.clip = anterior
                        }
                    }
                }
            }
        }

        texto("Mis Comisiones · ${formatoDia(ahora())}", 24, GRIS, false, 0)
        medir.dispose()

        val alto = (MARGEN * 2 + bloques.sumOf { it.alto }).coerceAtMost(ALTO_MAX)
        val bmp = BufferedImage(ANCHO, alto, BufferedImage.TYPE_INT_RGB)
        val g = bmp.createGraphics()
        suavizar(g)
        g.color = FONDO
        g.fillRect(0, 0, ANCHO, alto)
        g.color = ROSA
        g.fillRect(0, 0, ANCHO, 14)
        var y = MARGEN
        for (b in bloques) {
            if (y > alto) break
            b.dibujar(g, y)
            y += b.alto
        }
        g.dispose()

        val archivo = archivoDe(m.id)
        ImageIO.write(bmp, "png", archivo)
        return archivo
    }

fun diasDesde(t: Long?): Int {
    if (t == null || t <= 0L) return 0
    val z = ZoneId.systemDefault()
    val d = ChronoUnit.DAYS.between(Instant.ofEpochMilli(t).atZone(z).toLocalDate(), LocalDate.now(z))

}

object Traductor {
    fun traducir(texto: String): String {
        if (texto.isBlank()) return ""
        return texto.split("\n").joinToString("\n") { linea ->
            if (linea.isBlank()) linea else trozo(linea)
        }
    }

    private fun trozo(t: String): String {
        val q = URLEncoder.encode(t, "UTF-8")
        val url = URI("https://translate.googleapis.com/translate_a/single?client=gtx&sl=auto&tl=es&dt=t&q=$q").toURL()
        val con = url.openConnection() as HttpURLConnection
        con.connectTimeout = 10_000
        con.readTimeout = 20_000
        con.setRequestProperty("User-Agent", "Mozilla/5.0")
        val respuesta = con.inputStream.bufferedReader(Charsets.UTF_8).use { it.readText() }
        val partes = Json.parseToJsonElement(respuesta).jsonArray[0].jsonArray
        return partes.joinToString("") { it.jsonArray[0].jsonPrimitive.content }
    }
}

object Archivos {

        try {

                null
            } else {

                        val g = it.createGraphics()
                        g.drawImage(img, 0, 0, null)
                        g.dispose()
                    }
                }
                val dest = File(Almacen.carpetaRefs, UUID.randomUUID().toString() + ".png")
                ImageIO.write(bi, "png", dest)
                dest.absolutePath

object Imagenes {
    private val cache = object : LinkedHashMap<String, ImageBitmap>(64, 0.75f, true) {
        override fun removeEldestEntry(eldest: MutableMap.MutableEntry<String, ImageBitmap>?): Boolean = size > 80
    }

    fun cargarSkia(ruta: String): SkImage? =
        try {
            SkImage.makeFromEncoded(File(ruta).readBytes())
        } catch (e: Exception) {
            null
        }

    fun escalada(img: SkImage, anchoMax: Int): SkImage {
        if (img.width <= anchoMax || img.width <= 0) return img
        val h = maxOf(1, (img.height.toLong() * anchoMax / img.width).toInt())
        val s = Surface.makeRasterN32Premul(anchoMax, h)
        s.canvas.drawImageRect(
            img,
            Rect.makeWH(img.width.toFloat(), img.height.toFloat()),
            Rect.makeWH(anchoMax.toFloat(), h.toFloat()),
            SamplingMode.LINEAR,
            null,
            true
        )
        return s.makeImageSnapshot()
    }

    fun miniatura(ruta: String, anchoMax: Int): ImageBitmap? {
        val k = "$ruta@$anchoMax"
        synchronized(cache) { cache[k]?.let { return it } }
        val img = cargarSkia(ruta) ?: return null
        val bm = escalada(img, anchoMax).toComposeImageBitmap()
        synchronized(cache) { cache[k] = bm }
        return bm
    }

    fun sinCache(ruta: String): ImageBitmap? = cargarSkia(ruta)?.toComposeImageBitmap()

    fun olvidar(ruta: String) {
        synchronized(cache) { cache.keys.removeIf { it.startsWith("$ruta@") } }
    }

    fun aBuffered(img: SkImage): BufferedImage? {
        val data = img.encodeToData(EncodedImageFormat.PNG) ?: return null
        return ImageIO.read(ByteArrayInputStream(data.bytes))
    }
}

sealed interface Ruta {
    data object Clientes : Ruta
    data object Horario : Ruta
    data object Galeria : Ruta
    data object Etiquetas : Ruta
    data object Ajustes : Ruta
    data class Cliente(val id: Long) : Ruta
    data class Comision(val id: Long) : Ruta
    data class Ficha(val id: Long) : Ruta
    data class Etiqueta(val id: Long) : Ruta
}

class Nav {
    val pila = mutableStateListOf<Ruta>(Ruta.Clientes)
    val actual: Ruta get() = pila.last()
    fun ir(r: Ruta) {
        if (pila.last() != r) pila.add(r)
    }
    fun atras() {
        if (pila.size > 1) pila.removeAt(pila.lastIndex)
    }
    fun raiz(r: Ruta) {
        pila.clear()
        pila.add(r)
    }
}

// Visor de imagen a pantalla completa (se muestra encima de todo)
object Visor {
    var ruta by mutableStateOf<String?>(null)
    var alBorrar by mutableStateOf<(() -> Unit)?>(null)
    fun abrir(r: String, borrar: (() -> Unit)? = null) {
        alBorrar = borrar
        ruta = r
    }
    fun cerrar() {
        ruta = null
        alBorrar = null
    }
}

@Composable
private fun VisorImagen(ruta: String) {
    var escala by remember(ruta) { mutableStateOf(1f) }
    var desplazamiento by remember(ruta) { mutableStateOf(Offset.Zero) }
    val bm by produceState<ImageBitmap?>(null, ruta) {
        value = withContext(Dispatchers.IO) { Imagenes.miniatura(ruta, 2600) }
    }
    Box(
        Modifier
            .fillMaxSize()
            .background(Color.Black.copy(alpha = 0.94f))
            .clickable(interactionSource = remember { MutableInteractionSource() }, indication = null) { Visor.cerrar() }
    ) {
        val b = bm
        if (b != null) {
            androidx.compose.foundation.Image(
                b,
                contentDescription = null,
                contentScale = ContentScale.Fit,
                modifier = Modifier
                    .fillMaxSize()
                    .padding(24.dp)
                    .onPointerEvent(PointerEventType.Scroll) { ev ->
                        val dy = ev.changes.firstOrNull()?.scrollDelta?.y ?: 0f
                        escala = (escala * if (dy < 0) 1.15f else 1f / 1.15f).coerceIn(1f, 8f)
                        if (escala == 1f) desplazamiento = Offset.Zero
                    }
                    .pointerInput(ruta) {
                        detectDragGestures { cambio, arrastre ->
                            cambio.consume()
                            if (escala > 1f) desplazamiento += arrastre
                        }
                    }
                    .graphicsLayer {
                        scaleX = escala
                        scaleY = escala
                        translationX = desplazamiento.x
                        translationY = desplazamiento.y
                    }
            )
        } else {
            CircularProgressIndicator(Modifier.align(Alignment.Center))
        }
        Row(
            Modifier.align(Alignment.TopEnd).padding(12.dp),
            horizontalArrangement = Arrangement.spacedBy(8.dp)
        ) {
            FilledTonalButton(onClick = { Archivos.abrir(File(ruta)) }) {
                Icon(Icons.Default.OpenInNew, contentDescription = null, modifier = Modifier.size(18.dp))
                Spacer(Modifier.width(6.dp))
                Text("Abrir en Windows")
            }
            val borrar = Visor.alBorrar
            if (borrar != null) {
                FilledTonalButton(onClick = {
                    borrar()
                    Visor.cerrar()
                }) {
                    Icon(Icons.Default.Delete, contentDescription = null, modifier = Modifier.size(18.dp))
                    Spacer(Modifier.width(6.dp))
                    Text("Borrar")
                }
            }
            FilledTonalButton(onClick = { Visor.cerrar() }) {
                Icon(Icons.Default.Close, contentDescription = null, modifier = Modifier.size(18.dp))
                Spacer(Modifier.width(6.dp))
                Text("Cerrar (Esc)")
            }
        }
        Text(
            "Rueda del mouse: acercar/alejar · Arrastra para moverte",
            color = Color.White.copy(alpha = 0.6f),
            style = MaterialTheme.typography.labelMedium,
            modifier = Modifier.align(Alignment.BottomCenter).padding(12.dp)
        )
    }

    if (nueva) {
        DialogoCampos(
            "Nueva comisión",
            listOf(
                Campo("Título (ej. Miku en un parque)", obligatorio = true),
                Campo("Lo que pidió el cliente (pégalo en inglés)", multilinea = true)
            ),
            confirmar = "Crear",
            onDismiss = { nueva = false }
        ) { v ->
            nueva = false
            val nuevoId = Acciones.crearComision(id, v[0], v[1])
            nav.ir(Ruta.Comision(nuevoId))
        }
    }
    if (editar && c != null) DialogoEditarCliente(c) { editar = false }
    if (borrar && c != null) {
        DialogoConfirmar(
            "¿Eliminar a ${c.nombre}?",
            "Se borrarán también todas sus comisiones, referencias e imágenes. No se puede deshacer.",
            onDismiss = { borrar = false }
        ) {
            borrar = false
            Acciones.borrarCliente(c.id)
            nav.atras()
        }
    }
}

@Composable
fun TarjetaComision(m: Comision, diasRojo: Int, onClick: () -> Unit) {
    Card(onClick = onClick, modifier = Modifier.fillMaxWidth()) {
        Column(Modifier.padding(start = 4.dp, end = 12.dp, top = 8.dp, bottom = 6.dp)) {
            Row(verticalAlignment = Alignment.CenterVertically) {
                Checkbox(checked = m.terminada, onCheckedChange = { Acciones.setTerminada(m.id, it) })
                Column(Modifier.weight(1f)) {
                    Text(
                        m.titulo,
                        style = MaterialTheme.typography.titleMedium,
                        maxLines = 2,
                        overflow = TextOverflow.Ellipsis,
                        textDecoration = if (m.terminada) TextDecoration.LineThrough else null
                    )
                    val rec = m.recordatorio
                    if (rec != null) {
                        Row(verticalAlignment = Alignment.CenterVertically) {
                            Icon(
                                Icons.Default.Alarm, contentDescription = null,
                                tint = MaterialTheme.colorScheme.tertiary, modifier = Modifier.size(14.dp)
                            )
                            Spacer(Modifier.width(4.dp))
                            Text(formatoFecha(rec), style = MaterialTheme.typography.labelSmall, color = MaterialTheme.colorScheme.tertiary)
                        }
                    }
                }
                DiasBadge(m.ultimoTrabajo, m.terminada, diasRojo)
            }
            Spacer(Modifier.size(6.dp))
            Row(Modifier.padding(start = 8.dp)) {
                EtapasSelector(m.etapa, m.terminada, onElegir = { Acciones.setEtapa(m.id, it) }, compacto = true)
            }
            if (!m.terminada) {
                TextButton(onClick = { Acciones.dibujeHoy(m.id) }, modifier = Modifier.align(Alignment.End)) {
                    Icon(Icons.Default.Brush, contentDescription = null, modifier = Modifier.size(18.dp))
                    Spacer(Modifier.width(6.dp))
                    Text("Dibujé hoy")
                }
            }
        }
    }
}
